cmake_add_benchmark(
  bench_cnet_adapter
  SOURCES bench_cnet_adapter.c cnet_adapter_benchmark_stats.c
  FOLDER "io/cnet/benchmarks"
  LIBS TurboFlow::CNetAdapter Salts::TinyTest
  INCLUDES ${CMAKE_CURRENT_SOURCE_DIR})

find_package(Git REQUIRED)
execute_process(
  COMMAND "${GIT_EXECUTABLE}" rev-parse --verify HEAD
  WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
  RESULT_VARIABLE TURBO_FLOW_CNET_BENCHMARK_GIT_STATUS
  OUTPUT_VARIABLE TURBO_FLOW_CNET_BENCHMARK_GIT_COMMIT
  ERROR_VARIABLE TURBO_FLOW_CNET_BENCHMARK_GIT_ERROR
  OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT TURBO_FLOW_CNET_BENCHMARK_GIT_STATUS EQUAL 0 OR
   TURBO_FLOW_CNET_BENCHMARK_GIT_COMMIT STREQUAL "")
  message(FATAL_ERROR
          "bench_cnet_adapter requires an exact Git commit: ${TURBO_FLOW_CNET_BENCHMARK_GIT_ERROR}")
endif()
if(CMAKE_BUILD_TYPE STREQUAL "")
  message(FATAL_ERROR "bench_cnet_adapter requires a single-config CMAKE_BUILD_TYPE")
endif()
if(NOT DEFINED ENV{TURBO_FLOW_ACTIVE_PRESET} OR
   "$ENV{TURBO_FLOW_ACTIVE_PRESET}" STREQUAL "")
  message(FATAL_ERROR
          "bench_cnet_adapter requires a public user preset with TURBO_FLOW_ACTIVE_PRESET")
endif()
cmake_host_system_information(RESULT TURBO_FLOW_CNET_BENCHMARK_CPU_MODEL QUERY PROCESSOR_NAME)
string(STRIP "${TURBO_FLOW_CNET_BENCHMARK_CPU_MODEL}" TURBO_FLOW_CNET_BENCHMARK_CPU_MODEL)
if(TURBO_FLOW_CNET_BENCHMARK_CPU_MODEL STREQUAL "")
  message(FATAL_ERROR "bench_cnet_adapter requires a build-host CPU model")
endif()
string(REGEX REPLACE "[^A-Za-z0-9._-]" "_" TURBO_FLOW_CNET_BENCHMARK_CPU_MODEL
                     "${TURBO_FLOW_CNET_BENCHMARK_CPU_MODEL}")
target_compile_definitions(
  bench_cnet_adapter
  PRIVATE TF_CNET_BENCH_GIT_COMMIT="${TURBO_FLOW_CNET_BENCHMARK_GIT_COMMIT}"
          TF_CNET_BENCH_GIT_EXECUTABLE="${GIT_EXECUTABLE}"
          TF_CNET_BENCH_SOURCE_DIR="${PROJECT_SOURCE_DIR}"
          TF_CNET_BENCH_CPU_MODEL="${TURBO_FLOW_CNET_BENCHMARK_CPU_MODEL}"
          TF_CNET_BENCH_BUILD_TYPE="${CMAKE_BUILD_TYPE}"
          TF_CNET_BENCH_PRESET="$ENV{TURBO_FLOW_ACTIVE_PRESET}"
          TF_CNET_BENCH_ASAN=$<BOOL:${ENABLE_SANITIZER_ADDRESS}>)
