cmake_minimum_required(VERSION 3.20)
project(CHTTPClosureFixture LANGUAGES C)

if(NOT CHTTP_CLOSURE_SCENARIO MATCHES "^(legacy|wrong-crt|unresolved|outside)$")
  message(FATAL_ERROR "Unknown CHTTP_CLOSURE_SCENARIO: ${CHTTP_CLOSURE_SCENARIO}")
endif()

add_library(closure_leaf SHARED leaf.c)
add_library(closure_mid SHARED mid.c)
add_library(closure_root SHARED root.c)
target_link_libraries(closure_mid PRIVATE closure_leaf)
target_link_libraries(closure_root PRIVATE closure_mid)
if(CHTTP_CLOSURE_SCENARIO STREQUAL "legacy")
  set_target_properties(closure_leaf PROPERTIES OUTPUT_NAME "salts_chttp-2")
elseif(CHTTP_CLOSURE_SCENARIO STREQUAL "wrong-crt")
  set_target_properties(closure_leaf PROPERTIES OUTPUT_NAME "vcruntime140d")
elseif(CHTTP_CLOSURE_SCENARIO STREQUAL "outside")
  set_target_properties(closure_leaf PROPERTIES OUTPUT_NAME "chttp_outside")
endif()

install(TARGETS closure_root closure_mid RUNTIME DESTINATION allowed/bin)
if(CHTTP_CLOSURE_SCENARIO STREQUAL "outside")
  install(TARGETS closure_leaf RUNTIME DESTINATION outside/bin)
elseif(NOT CHTTP_CLOSURE_SCENARIO STREQUAL "unresolved")
  install(TARGETS closure_leaf RUNTIME DESTINATION allowed/bin)
endif()
