cmake_minimum_required(VERSION 3.20)
project(CHTTPClosureFixture LANGUAGES C)

if(NOT CHTTP_CLOSURE_SCENARIO MATCHES "^(positive|debug-positive|legacy|wrong-crt|wrong-msvcp-crt|debug-release-crt|unresolved|outside|arbitrary-outside|system-forbidden|system-unsigned|system-signed-non-os)$")
  message(FATAL_ERROR "Unknown CHTTP_CLOSURE_SCENARIO: ${CHTTP_CLOSURE_SCENARIO}")
endif()

add_library(closure_leaf SHARED leaf.c)
add_library(closure_mid SHARED mid.c)
add_library(closure_root SHARED root.c)
target_link_libraries(closure_mid PRIVATE closure_leaf)
target_link_libraries(closure_root PRIVATE closure_mid)
if(CHTTP_CLOSURE_SCENARIO STREQUAL "legacy")
  set_target_properties(closure_leaf PROPERTIES OUTPUT_NAME "salts_chttp-2")
elseif(CHTTP_CLOSURE_SCENARIO STREQUAL "wrong-crt")
  set_target_properties(closure_leaf PROPERTIES OUTPUT_NAME "vcruntime140d")
elseif(CHTTP_CLOSURE_SCENARIO STREQUAL "wrong-msvcp-crt")
  set_target_properties(closure_leaf PROPERTIES OUTPUT_NAME "msvcp140d")
elseif(CHTTP_CLOSURE_SCENARIO STREQUAL "debug-release-crt")
  set_target_properties(closure_leaf PROPERTIES OUTPUT_NAME "vcruntime140")
elseif(CHTTP_CLOSURE_SCENARIO STREQUAL "outside")
  set_target_properties(closure_leaf PROPERTIES OUTPUT_NAME "chttp_outside")
elseif(CHTTP_CLOSURE_SCENARIO STREQUAL "arbitrary-outside")
  set_target_properties(closure_leaf PROPERTIES OUTPUT_NAME "vendor_helper")
elseif(CHTTP_CLOSURE_SCENARIO STREQUAL "system-forbidden")
  set_target_properties(closure_leaf PROPERTIES OUTPUT_NAME "salts_chttp_system")
elseif(CHTTP_CLOSURE_SCENARIO STREQUAL "system-unsigned")
  set_target_properties(closure_leaf PROPERTIES OUTPUT_NAME "unsigned_helper")
elseif(CHTTP_CLOSURE_SCENARIO STREQUAL "system-signed-non-os")
  set_target_properties(closure_leaf PROPERTIES OUTPUT_NAME "concrt140")
endif()

install(TARGETS closure_root closure_mid RUNTIME DESTINATION allowed/bin)
if(CHTTP_CLOSURE_SCENARIO MATCHES "^(outside|arbitrary-outside)$")
  install(TARGETS closure_leaf RUNTIME DESTINATION outside/bin)
elseif(CHTTP_CLOSURE_SCENARIO MATCHES "^(system-forbidden|system-unsigned)$")
  install(TARGETS closure_leaf RUNTIME DESTINATION system/bin)
elseif(NOT CHTTP_CLOSURE_SCENARIO MATCHES "^(unresolved|system-signed-non-os)$")
  install(TARGETS closure_leaf RUNTIME DESTINATION allowed/bin)
endif()
